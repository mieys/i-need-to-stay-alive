class ShardIcon extends Control:
	var dim: bool = false

	func _init(px: float = 32.0, dimmed: bool = false) -> void:
		custom_minimum_size = Vector2(px, px)
		mouse_filter = Control.MOUSE_FILTER_IGNORE
		dim = dimmed

	func _draw() -> void:
		var s: float = size.x
		var c: float = s * 0.5
		var top := Vector2(c, s * 0.02)
		var mid_y: float = s * 0.38
		var l := Vector2(s * 0.10, mid_y)
		var r := Vector2(s * 0.90, mid_y)
		var bot := Vector2(c, s * 0.98)
		var m := Vector2(c, mid_y)
		var k: float = 0.55 if dim else 1.0
		var tint := Color(k, k, k, 1.0)
		draw_colored_polygon(PackedVector2Array([top, m, l]), Color("#a8f0f4") * tint)
		draw_colored_polygon(PackedVector2Array([top, r, m]), Color("#6fd0da") * tint)
		draw_colored_polygon(PackedVector2Array([l, m, bot]), Color("#3f9fb0") * tint)
		draw_colored_polygon(PackedVector2Array([m, r, bot]), Color("#2a7a8c") * tint)
		draw_polyline(PackedVector2Array([top, r, bot, l, top]), Color("#4a2c1a"), maxf(2.0, s / 12.0))

static func _load_layout() -> String:
	var cfg := ConfigFile.new()
	if cfg.load(UI_CFG) == OK:
		var v: String = str(cfg.get_value("ui", "layout", "a"))
		for pair: Array in LAYOUTS:
			if pair[0] == v:
				return v
	return "a"


func _save_layout() -> void:
	var cfg := ConfigFile.new()
	cfg.set_value("ui", "layout", _layout)
	cfg.save(UI_CFG)

## İçeriği baştan kurar (satın alma / seçim / onay durumu değişince). Kaydırma konumu ve odaktaki düğme korunur.
func _rebuild_list(reset_scroll: bool = false) -> void:
	var keep: int = 0
	if not reset_scroll and is_instance_valid(_scroll):
		keep = _scroll.scroll_vertical
	var focus_sid: String = ""
	var fo: Control = get_viewport().gui_get_focus_owner() if is_inside_tree() else null
	if fo != null and fo.has_meta("sid"):
		focus_sid = str(fo.get_meta("sid"))
	_scroll = null
	_list = null
	for c in _content.get_children():
		_content.remove_child(c)
		c.queue_free()
	_balance_label.text = str(MetaProgressScript.shards())
	for key: String in _tab_buttons.keys():
		MenuKit.style_button(_tab_buttons[key] as Button, "sage" if key == _tab else "tan", MenuKit.FS_BODY)
	for key: String in _layout_buttons.keys():
		MenuKit.style_button(_layout_buttons[key] as Button, "sage" if key == _layout else "tan", MenuKit.FS_SMALL)
	_layout_row.visible = _tab == TAB_UNLOCKS
	if _tab == TAB_UPGRADES:
		_build_upgrades_tab()
	else:
		var items: Array = _items()
		_ensure_selection(items)
		var view: Control = call("_view_" + _layout, items)
		view.size_flags_vertical = Control.SIZE_EXPAND_FILL
		_content.add_child(view)
	UISound.connect_all_buttons(self)
	_restore_view.call_deferred(keep, focus_sid)


func _restore_view(scroll_v: int, focus_sid: String) -> void:
	if is_instance_valid(_scroll):
		_scroll.scroll_vertical = scroll_v
	if focus_sid == "":
		return
	for b: Node in find_children("*", "Button", true, false):
		if b.has_meta("sid") and str(b.get_meta("sid")) == focus_sid and not (b as Button).disabled:
			(b as Button).grab_focus()
			return


func _on_tab_pressed(tab: String) -> void:
	if tab == _tab:
		return
	_tab = tab
	_confirm_id = ""
	_rebuild_list(true)


func _on_layout_pressed(l: String) -> void:
	if l == _layout:
		return
	_layout = l
	_sel_kind = ""
	_sel_key = null
	_confirm_id = ""
	_save_layout()
	_rebuild_list(true)

## Satır bilgisi: {"name", "desc", "icon"} - her tür kendi kataloğundan.
func _info(kind: String, key: Variant) -> Dictionary:
	match kind:
		MetaProgressScript.CHAR:
			var def: Dictionary = Characters.get_def(int(key))
			var skills: Array = []
			for pair: Array in [["skill_name", "Q"], ["skill2_name", "E"], ["skill3_name", "R"]]:
				var sn: String = str(def.get(pair[0], ""))
				if sn != "":
					skills.append("%s: %s" % [pair[1], sn])
			return {"name": str(def.get("name", key)), "desc": "  ".join(skills),
					"icon": LobbyUI.face_texture(str(def.get("portrait", "")))}
		MetaProgressScript.WEAPON:
			var wk: String = str(key)
			return {"name": WeaponCatalog.display_name(wk), "desc": "%s - %s" % [WeaponCatalog.category_label(wk), WeaponCatalog.trait_text(wk)],
					"icon": WeaponCatalog.icon(wk)}
		MetaProgressScript.PET:
			var pk: String = str(key)
			return {"name": PetCatalog.display_name(pk), "desc": PetCatalog.description(pk), "icon": PetCatalog.icon(pk)}
		MetaProgressScript.SPIRIT:
			var sd: Dictionary = SpiritualSkillsScript.get_def(str(key))
			var tex_path: String = str(sd.get("icon", ""))
			return {"name": str(sd.get("name", key)), "desc": str(sd.get("desc", "")),
					"icon": load(tex_path) as Texture2D if tex_path != "" and ResourceLoader.exists(tex_path) else null}
	return {"name": str(key)}


## Tüm kilitli/açık katalog öğeleri (tür sırası, ucuzdan pahalıya).
func _items() -> Array:
	var out: Array = []
	for it: Dictionary in MetaProgressScript.locked_items():
		var info: Dictionary = _info(it["kind"], it["key"])
		out.append({"kind": it["kind"], "key": it["key"], "cost": int(it["cost"]), "name": str(info.get("name", it["key"])),
				"desc": str(info.get("desc", "")), "icon": info.get("icon", null),
				"open": MetaProgressScript.is_unlocked(it["kind"], it["key"]), "can": MetaProgressScript.can_buy(it["kind"], it["key"])})
	return out


func _of_kind(items: Array, kind: String) -> Array:
	return items.filter(func(i: Dictionary) -> bool: return i["kind"] == kind)


func _id(it: Dictionary) -> String:
	return "%s:%s" % [it["kind"], str(it["key"])]


func _state(it: Dictionary) -> String:
	return "open" if it["open"] else ("can" if it["can"] else "no")


func _need(it: Dictionary) -> int:
	return maxi(0, int(it["cost"]) - MetaProgressScript.shards())


## Alınabilir -> alınamayan (ucuzdan pahalıya) -> açık olanlar.
func _sorted(items: Array) -> Array:
	var a: Array = items.duplicate()
	a.sort_custom(func(x: Dictionary, y: Dictionary) -> bool:
		var gx: int = 2 if x["open"] else (0 if x["can"] else 1)
		var gy: int = 2 if y["open"] else (0 if y["can"] else 1)
		if gx != gy:
			return gx < gy
		return int(x["cost"]) < int(y["cost"]))
	return a


func _open_count(items: Array) -> int:
	var n: int = 0
	for i: Dictionary in items:
		if i["open"]:
			n += 1
	return n


func _find_sel(items: Array) -> Dictionary:
	if _sel_kind == "":
		return {}
	for i: Dictionary in items:
		if i["kind"] == _sel_kind and str(i["key"]) == str(_sel_key):
			return i
	return {}


## Seçili öğe geçerli mi: A seçili kategoriden, C ilk (en alınabilir) öğeden başlar; D'de seçim yoksa "sıradaki hedef" gösterilir.
func _ensure_selection(items: Array) -> void:
	var cur: Dictionary = _find_sel(items)
	if _layout == "d":
		if cur.is_empty():
			_sel_kind = ""
			_sel_key = null
		return
	if _layout == "a" and (cur.is_empty() or cur["kind"] != _cat):
		cur = (_sorted(_of_kind(items, _cat)) as Array)[0]
	elif cur.is_empty():
		cur = (_sorted(items) as Array)[0]
	_sel_kind = str(cur["kind"])
	_sel_key = cur["key"]


func _select(it: Dictionary) -> void:
	_sel_kind = str(it["kind"])
	_sel_key = it["key"]
	_confirm_id = ""
	_rebuild_list()


func _set_cat(kind: String) -> void:
	_cat = kind
	_sel_kind = ""
	_confirm_id = ""
	_rebuild_list(true)


func _on_buy_pressed(kind: String, key: Variant) -> void:
	var id: String = "%s:%s" % [kind, str(key)]
	if _confirm_id != id:
		_confirm_id = id
		_rebuild_list()
		return
	_confirm_id = ""
	MetaProgressScript.buy(kind, key)
	_rebuild_list()

## Satın alma düğmesi: açık = pasif "AÇIK", alınabilir = "AÇ" (ikinci basışta "ONAYLA?"), yetmiyor = pasif "N EKSİK".
func _buy_btn(it: Dictionary, h: float = 56.0, fs: int = MenuKit.FS_BODY, with_price: bool = false) -> Button:
	var st: String = _state(it)
	var id: String = _id(it)
	var b: Button
	if st == "open":
		b = MenuKit.make_button("AÇIK", "sage", fs, h)
		b.disabled = true
	elif st == "can":
		var asking: bool = _confirm_id == id
		var label: String = "ONAYLA?" if asking else ("AÇ  -  %d" % int(it["cost"]) if with_price else "AÇ")
		b = MenuKit.make_button(label, "rose" if asking else "sage", fs, h)
		b.pressed.connect(_on_buy_pressed.bind(it["kind"], it["key"]))
	else:
		b = MenuKit.make_button("%d EKSİK" % _need(it), "tan", fs, h)
		b.disabled = true
	_tag_sid(b, "buy:" + id)
	return b

func _make_upgrade_row(d: Dictionary) -> Control:
	var id: String = str(d["id"])
	var lv: int = MetaUpgradesScript.level(id)
	var mx: int = int(d["max"])
	var maxed: bool = lv >= mx
	var per: float = float(d["per"])
	var row := HBoxContainer.new()
	row.add_theme_constant_override("separation", 12)

	var icon := TextureRect.new()
	icon.custom_minimum_size = Vector2(ROW_ICON, ROW_ICON)
	icon.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	icon.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_CENTERED
	icon.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
	var icon_key: String = str(d.get("icon", ""))
	var icon_file: String = str(StatIconScript.ICON_PATHS.get(icon_key, ""))
	if icon_file != "" and ResourceLoader.exists(StatIconScript.ICON_DIR + icon_file):
		icon.texture = load(StatIconScript.ICON_DIR + icon_file) as Texture2D
	row.add_child(icon)

	var col := VBoxContainer.new()
	col.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	col.add_theme_constant_override("separation", 0)
	row.add_child(col)
	col.add_child(MenuKit.make_label("%s   %d / %d" % [str(d["name"]), lv, mx], MenuKit.FS_BODY, MenuKit.C_GOOD if maxed else MenuKit.C_TEXT))
	var effect: String = "şu an %s" % MetaUpgradesScript.format_value(id, per * float(lv))
	if not maxed:
		effect += "  ->  sonraki %s" % MetaUpgradesScript.format_value(id, per * float(lv + 1))
	col.add_child(_desc_label("%s  (%s)" % [str(d["desc"]), effect], MenuKit.FS_SMALL))

	var side := VBoxContainer.new()
	side.alignment = BoxContainer.ALIGNMENT_CENTER
	side.custom_minimum_size.x = 150.0
	row.add_child(side)
	var btn: Button
	if maxed:
		btn = MenuKit.make_button("TAM", "sage", MenuKit.FS_SMALL, 44)
		btn.disabled = true
	else:
		var can: bool = MetaUpgradesScript.can_buy(id)
		side.add_child(_price(MetaUpgradesScript.next_cost(id), 28, C_PRICE if can else MenuKit.C_BAD, MenuKit.FS_BODY, not can))
		btn = MenuKit.make_button("YÜKSELT", "sage" if can else "tan", MenuKit.FS_SMALL, 44)
		btn.disabled = not can
		btn.pressed.connect(_on_upgrade_pressed.bind(id))
	_tag_sid(btn, "up:" + id)
	side.add_child(btn)
	return row


func _make_reset_row() -> Control:
	var panel := MenuKit.make_panel("inset")
	var row := HBoxContainer.new()
	row.add_theme_constant_override("separation", 12)
	panel.add_child(row)
	var col := VBoxContainer.new()
	col.size_flags_horizontal = Control.SIZE_EXPAND_FILL
	row.add_child(col)
	col.add_child(MenuKit.make_label("Geliştirmeleri Sıfırla", MenuKit.FS_BODY, MenuKit.C_TEXT))
	var info: String = "Sıfırlanacak geliştirme yok."
	if MetaUpgradesScript.can_reset():
		info = "Harcanan %d Ruh Kıymığı, %d ücret kesilip %d olarak iade edilir." % [
				MetaUpgradesScript.total_spent(), MetaUpgradesScript.reset_fee(), MetaUpgradesScript.reset_refund()]
	col.add_child(_desc_label(info, MenuKit.FS_SMALL))
	var asking: bool = _confirm_id == "reset"
	var btn: Button = MenuKit.make_button("ONAYLA?" if asking else "SIFIRLA", "rose" if asking else "tan", MenuKit.FS_BODY, 52)
	btn.disabled = not MetaUpgradesScript.can_reset()
	btn.custom_minimum_size.x = 200.0
	btn.pressed.connect(_on_reset_pressed)
	_tag_sid(btn, "reset")
	row.add_child(btn)
	return panel


func _on_upgrade_pressed(id: String) -> void:
	_confirm_id = ""
	MetaUpgradesScript.buy(id)
	_rebuild_list()


func _on_reset_pressed() -> void:
	if _confirm_id != "reset":
		_confirm_id = "reset"
		_rebuild_list()
		return
	_confirm_id = ""
	MetaUpgradesScript.reset_all()
	_rebuild_list()


func _unhandled_input(event: InputEvent) -> void:
	if event.is_action_pressed("ui_cancel"):
		_on_close_pressed()
		get_viewport().set_input_as_handled()


func _on_close_pressed() -> void:
	closed.emit()
	queue_free()
